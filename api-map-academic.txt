GET /api/academic/dashboard
   returns: .json({ success: true, message: 'Academic dashboard fetched.', data: { currentAcademicYear: currentYear, currentTerm, totalClasses, totalSections, totalSubjects, totalRooms, totalStudents, totalTeachers, sectionsByGrade, }, }); });

GET /api/academic/years/current
   returns: .json({ success: true, message: 'Current academic year fetched.', data: { academicYear: { ...academicYear.toObject(), terms } }, }); }); // ═══════════════════════════════════════════ // TERM // ═══════════════════════════════════════════

GET /api/academic/years
   returns: .json({ success: true, message: 'Academic years fetched.', data: { academicYears }, }); });

POST /api/academic/years   roles: ...adminRoles
   body: name, startDate, endDate, isCurrent, description
   returns: .json({ success: true, message: 'Academic year created successfully.', data: { academicYear }, }); });

GET /api/academic/years/:id
   returns: .json({ success: true, message: 'Academic year fetched.', data: { academicYear }, }); });

PATCH /api/academic/years/:id   roles: ...adminRoles
   body: isCurrent
   returns: .json({ success: true, message: 'Academic year updated.', data: { academicYear }, }); });

DELETE /api/academic/years/:id   roles: ...adminRoles
   returns: .json({ success: true, message: 'Academic year deleted.', }); });
