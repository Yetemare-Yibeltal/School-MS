
##### /api/students   (student.routes.js)

GET /api/students/stats   roles: 'super_admin', 'admin', 'teacher', 'receptionist'
   returns: .json({ success: true, message: 'Student dashboard stats fetched.', data: { totalStudents, activeStudents, inactiveStudents: totalStudents - activeStudents, suspendedStudents, byGrade, byGender, recentAdmissions, }, }); }); // ─── Bulk Import Students ─────────────────────

POST /api/students/bulk-import   roles: 'super_admin', 'admin', 'receptionist'
   body: students
   returns: .json({ success: true, message: `Imported ${results.success.length} students. ${results.errors.length} failed.`, data: results, }); }); // ─── Promote Students ─────────────────────────

POST /api/students/promote   roles: 'super_admin', 'admin'
   body: fromGrade, toGrade, fromAcademicYear, toAcademicYear, studentIds
   returns: .json({ success: true, message: `${result.modifiedCount} students promoted from ${fromGrade} to ${toGrade}.`, data: { promoted: result.modifiedCount }, }); });

GET /api/students   roles: 'super_admin', 'admin', 'teacher', 'receptionist', 'accountant', 'librarian'
   query: grade, section, gender, status = 'active', academicYear, search, page = 1, limit = 20, sort = 'firstName', order = 'asc', startDate, endDate,
   returns: .json({ success: true, message: 'Students fetched successfully.', data: { students }, pagination: { total, page: parseInt(page), limit: parseInt(limit), totalPages, hasNextPage: parseInt(page) < totalPages, hasPreviousPage: parseInt(page) > 1, }, }); }); // ─── Get Student by ID ────────────────────────
   (paginated)

POST /api/students   roles: 'super_admin', 'admin', 'receptionist'
   body: firstName, fatherName, grandFatherName, firstNameAmharic, fatherNameAmharic, grandFatherNameAmharic, gender, dateOfBirth, nationality, religion, bloodGroup, grade, section, academicYear, admissionDate, admissionType, previousSchool, previousGrade, address, phone, email, medicalInfo, primaryGuardian, notes,
   returns: .json({ success: true, message: 'Student created successfully.', data: { student: { ...student.toObject(), guardian, }, }, }); }); // ─── Get All Students ─────────────────────────

GET /api/students/:id
   returns: .json({ success: true, message: 'Student fetched.', data: { student }, }); }); // ─── Update Student ───────────────────────────

PATCH /api/students/:id   roles: 'super_admin', 'admin', 'receptionist', 'teacher'
   returns: .json({ success: true, message: 'Student updated successfully.', data: { student }, }); }); // ─── Delete Student ───────────────────────────

DELETE /api/students/:id   roles: 'super_admin', 'admin'
   returns: .json({ success: true, message: 'Student deactivated successfully.', }); }); // ─── Transfer Student ─────────────────────────

POST /api/students/:id/photo   roles: 'super_admin', 'admin', 'receptionist'
   returns: .json({ success: true, message: 'Photo uploaded successfully.', data: { photo: student.photo }, }); }); // ─── Get Student Attendance Summary ───────────

POST /api/students/:id/transfer   roles: 'super_admin', 'admin'
   body: newGrade, newSection, newAcademicYear, transferReason
   returns: .json({ success: true, message: 'Student transferred successfully.', data: { student }, }); }); // ─── Upload Student Photo ─────────────────────

GET /api/students/:id/attendance
   query: academicYear, term
   returns: .json({ success: true, message: 'Attendance summary fetched.', data: { stats, recentAttendance, }, }); }); // ─── Get Student Fee Summary ──────────────────

GET /api/students/:id/fees
   query: academicYear
   returns: .json({ success: true, message: 'Fee summary fetched.', data: feeSummary, }); }); // ─── Get Student Exam Results ─────────────────

GET /api/students/:id/results
   query: academicYear, term
   returns: .json({ success: true, message: 'Exam results fetched.', data: { results }, }); }); // ─── Get Student Suspension History ───────────

GET /api/students/:id/suspensions   roles: 'super_admin', 'admin', 'teacher'
   returns: .json({ success: true, message: 'Suspension history fetched.', data: { suspensions }, }); }); // ─── Get Student Library History ──────────────
