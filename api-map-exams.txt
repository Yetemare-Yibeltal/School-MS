
##### /api/exams   (exam.routes.js)

GET /api/exams/dashboard   roles: ...teacherRoles, 'receptionist'
   returns: .json({ success: true, message: 'Exam dashboard fetched.', data: { currentAcademicYear: currentYear, currentTerm, totalExams, scheduled, completed, ongoing, recentResults, examsByGrade, }, }); });

GET /api/exams/types
   returns: .json({ success: true, message: 'Exam types fetched.', data: { examTypes }, }); });

POST /api/exams/types   roles: ...adminRoles
   body: name, code, description, totalMarks, isCA, isTerminal, isMandatory, sortOrder
   returns: .json({ success: true, message: 'Exam type created.', data: { examType }, }); });

PATCH /api/exams/types/:id   roles: ...adminRoles
   returns: .json({ success: true, message: 'Exam type updated.', data: { examType }, }); });

DELETE /api/exams/types/:id   roles: ...adminRoles
   returns: .json({ success: true, message: 'Exam type deactivated.', }); }); // ═══════════════════════════════════════════ // EXAM // ═══════════════════════════════════════════

GET /api/exams   roles: ...teacherRoles, 'student', 'parent'
   query: academicYear, term, grade, section, subject, examType, status, search, page = 1, limit = 20,
   returns: .json({ success: true, message: 'Exams fetched.', data: { exams }, pagination: { total, page: parseInt(page), totalPages: Math.ceil(total / parseInt(limit)), }, }); });
   (paginated)

POST /api/exams   roles: ...teacherRoles
   body: title, examType, academicYear, term, grade, section, subject, startDate, endDate, startTime, endTime, duration, room, invigilator, totalMarks, passingMarks, instructions, isOnline,
   returns: .json({ success: true, message: 'Exam scheduled successfully.', data: { exam }, }); });

GET /api/exams/:id
   returns: .json({ success: true, message: 'Exam fetched.', data: { exam, resultStats: resultStats[0] || null }, }); });

PATCH /api/exams/:id   roles: ...teacherRoles

PATCH /api/exams/:id/status   roles: ...teacherRoles

GET /api/exams/:examId/results   roles: ...teacherRoles, 'receptionist'
   query: page = 1, limit = 50
   returns: .json({ success: true, message: 'Exam results fetched.', data: { exam, results: rankedResults, stats, }, pagination: { total, page: parseInt(page), totalPages: Math.ceil(total / parseInt(limit)), }, }); });
   (paginated)

POST /api/exams/results   roles: ...teacherRoles
