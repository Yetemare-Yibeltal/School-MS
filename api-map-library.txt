GET /api/library/books
   query: category, language, grade, isAvailable, isReference, acquisitionSource, search, page = 1, limit = 20, sort = 'title', order = 'asc',
   returns: .json({ success: true, message: 'Books fetched.', data: { books }, pagination: { total, page: parseInt(page), limit: parseInt(limit), totalPages: Math.ceil(total / parseInt(limit)), hasNextPage: parseInt(page) < Math.ceil(total / parseInt(limit)), hasPreviousPage: parseInt(page) > 1, }, }); });
   (paginated)
